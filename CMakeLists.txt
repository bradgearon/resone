cmake_minimum_required(VERSION 3.25)
project(WdsResoneCore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(resone_audio native/src/AudioEngine.cpp)
target_include_directories(resone_audio PUBLIC native/include native/vendor)
target_link_libraries(resone_audio PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
add_executable(resone_audio_test tests/audio_core.cpp)
target_link_libraries(resone_audio_test PRIVATE resone_audio)
enable_testing()
add_test(NAME audio_core COMMAND resone_audio_test ${CMAKE_CURRENT_SOURCE_DIR})
