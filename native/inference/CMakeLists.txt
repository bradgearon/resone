cmake_minimum_required(VERSION 3.25)
project(resone_inference LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT LLAMA_CPP_DIR)
 message(FATAL_ERROR "Set LLAMA_CPP_DIR to the reviewed, pinned llama.cpp source used for this engine pack")
endif()
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(${LLAMA_CPP_DIR} llama)
add_library(resone_inference SHARED resone_inference.cpp)
target_include_directories(resone_inference PRIVATE ${LLAMA_CPP_DIR}/common)
target_link_libraries(resone_inference PRIVATE llama llama-common)
install(TARGETS resone_inference RUNTIME DESTINATION . COMPONENT ResoneInference LIBRARY DESTINATION . COMPONENT ResoneInference)
